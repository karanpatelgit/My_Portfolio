Code Snapshot

A peek at the file organiser script logic.

Python 3.x os & shutil
import os, shutil FOLDER_RULES = { "PDFs": [".pdf"], "Images": [".jpg", ".jpeg", ".png", ".gif", ".webp"], "Videos": [".mp4", ".mkv", ".avi"], "Code": [".py", ".js", ".html", ".css", ".java"], "Docs": [".docx", ".xlsx", ".pptx", ".txt"], } def organise(path): for file in os.listdir(path): if os.path.isfile(os.path.join(path, file)): ext = os.path.splitext(file)[1].lower() for folder, exts in FOLDER_RULES.items(): if ext in exts: dest = os.path.join(path, folder) os.makedirs(dest, exist_ok=True) shutil.move(os.path.join(path, file), dest) break organise(r"C:\Users\Karan\Downloads")
